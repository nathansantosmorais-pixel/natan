LARGURA - 50: self.vel_x *= -0.95 if self.y - self.raio < 50 or self.y + self.raio > ALTURA - 50: self.vel_y *= -0.95 def distancia(b1, b2): return math.hypot(b1.x - b2.x, b1.y - b2.y) def colisao_bolas(b1, b2): if distancia(b1, b2) < b1.raio + b2.raio: dx = b2.x - b1.x dy = b2.y - b1.y dist = distancia(b1, b2) if dist == 0: return nx = dx / dist ny = dy / dist # Velocidade relativa dvx = b2.vel_x - b1.vel_x dvy = b2.vel_y - b1.vel_y # Componente da velocidade ao longo da normal p = 2 * (dvx * nx + dvy * ny) / (b1.mass + b2.mass) b1.vel_x += p * b1.mass * nx b1.vel_y += p * b1.mass * ny b2.vel_x -= p * b2.mass * nx b2.vel_y -= p * b2.mass * ny # Criar bolas bolas = [] # Bola branca (taco) branca = Bola(300, 300, BRANCO) bolas.append(branca) # Bolas coloridas cores = [VERMELHO, AZUL, AMARELO, (255, 0, 255), (0, 255, 255)] for i in range(5): bola = Bola(650 + i*40, 250 + (i%2)*50, cores[i%5]) bolas.append(bola) # Variáveis do taco taco_puxado = False forca = 0 angulo = 0 rodando = True while rodando: tela.fill((20, 20, 20)) # Desenhar mesa pygame.draw.rect(tela, VERDE_MESA, (50, 50, LARGURA-100, ALTURA-100)) pygame.draw.rect(tela, MARROM, (40, 40, LARGURA-80, ALTURA-80), 20) for evento in pygame.event.get(): if evento.type == pygame.QUIT: rodando = False if evento.type == pygame.MOUSEBUTTONDOWN and all(abs(b.vel_x) < 0.5 and abs(b.vel_y) < 0.5 for b in bolas): taco_puxado = True mx, my = pygame.mouse.get_pos() angulo = math.atan2(my -
