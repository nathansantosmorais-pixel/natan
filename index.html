function iniciarSaida(numeroVaga) { const index = vagasOcupadas.findIndex(v => v.vaga === numeroVaga); if (index === -1) return; const veiculo = vagasOcupadas[index]; const saidaDate = new Date(); const diffMs = saidaDate - veiculo.entradaDate; const minutos = Math.max(1, Math.ceil(diffMs / 60000)); const valor = calcularValor(minutos); const tempoFormatado = formatarTempo(minutos); const saidaHora = saidaDate.toLocaleTimeString('pt-BR', { hour: '2-digit', minute: '2-digit' }); veiculoSaindo = { ...veiculo, index, saidaHora, minutos, valor, tempoFormatado }; document.getElementById('conteudoTicket').innerHTML = `

TICKET DE ESTACIONAMENTO

Estacionamento Digital
CNPJ: 00.000.000/0001-00
Cliente: ${veiculo.pessoa}
CPF: ${veiculo.cpf}
Veículo: ${veiculo.veiculo}
Placa: ${veiculo.placa}
GR: ${veiculo.gr}
Entrada: ${veiculo.entrada}
Saída: ${saidaHora}
Tempo: ${tempoFormatado}
TOTAL A PAGAR R$ ${valor.toFixed(2).replace('.', ',')}
${dataHoraAtual()}
`; document.getElementById('modalTicket').classList.add('ativo'); } function fecharTicket() { document.getElementById('modalTicket').classList.remove('ativo'); veiculoSaindo = null; }
